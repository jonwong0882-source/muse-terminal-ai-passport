cmake_minimum_required(VERSION 3.16)
project(muse_sim C CXX)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LVGL_DIR "${CMAKE_CURRENT_LIST_DIR}/../../managed_components/lvgl__lvgl" CACHE PATH "Pinned LVGL source")
add_subdirectory("${LVGL_DIR}" lvgl)
set(MUSE_AVATAR_PATH "${CMAKE_CURRENT_LIST_DIR}/../../assets/images/muse-avatar-rgb565.bin")
configure_file(avatar_path.h.in avatar_path.h @ONLY)
add_executable(muse_sim main.c avatar_host.c ../../main/muse_ui.c ../../assets/fonts/muse_font_16.c)
target_include_directories(muse_sim PRIVATE ../../main ../bsp_stubs ../../components/bsp/include ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(muse_sim PRIVATE lvgl m)
